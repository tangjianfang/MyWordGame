```csharp
[Test]
public void SaveDirty_ConcurrentWriters_BothChunksPersisted()
{
    string dir = NewTempDir(); // 照本文件既有临时目录 helper
    for (int round = 0; round < 20; round++)
    {
        var worldA = new World(); var worldB = new World();
        var posA = new ChunkPos(round * 2, 0); var posB = new ChunkPos(round * 2 + 1, 0);
        worldA.SetBlock(4, 70, 4, BlockIds.Stone); worldA.SetBlock(5, 70, 5, BlockIds.Stone);
        worldB.SetBlock(4, 71, 4, BlockIds.Dirt);
        var a = System.Threading.Tasks.Task.Run(() => RegionSaveCoordinator.SaveDirty(worldA, dir));
        var b = System.Threading.Tasks.Task.Run(() => RegionSaveCoordinator.SaveDirty(worldB, dir));
        System.Threading.Tasks.Task.WaitAll(a, b);
        var check = new World();
        // 两个线程写的区块都要能读回（无锁时 read-modify-write 交错会丢一侧）
        RegionSaveCoordinator.TryLoadChunk(check, posA, dir);
        RegionSaveCoordinator.TryLoadChunk(check, posB, dir);
        Assert.That(check.GetBlock(5, 70, 5), Is.EqualTo(BlockIds.Stone),
            $"第 {round} 轮：A 侧区块丢失（双写者竞态）");
        Assert.That(check.GetBlock(4, 71, 4), Is.EqualTo(BlockIds.Dirt),
            $"第 {round} 轮：B 侧区块丢失（双写者竞态）");
    }
}
```

（写入侧用 `World.SetBlock` 制造脏块后直接 `SaveDirty(world, dir)` 同步重载——两个 `World` 实例各持各的脏块，落盘目标是同一目录，与线上「后台快照写 vs 卸载主线程写」同构。）
